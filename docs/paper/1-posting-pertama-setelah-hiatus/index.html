struct Record<'a> {
    name: &'a str,
    score: f64,
}

fn main() {
    let mut data = [
        Record { name: "Alice", score: 85.0 },
        Record { name: "Bob", score: 92.0 },
        Record { name: "Charlie", score: 78.0 },
        Record { name: "Diana", score: 88.0 },
        Record { name: "Evan", score: 95.0 },
    ];

    println!("Original List:");
    for r in &data {
        println!("{} - {:.0}", r.name, r.score);
    }

    data.sort_unstable_by(|a, b| {
        b.score.total_cmp(&a.score)
    });

    println!("\nSorted by Score (Descending):");
    for r in &data {
        println!("{} - {:.0}", r.name, r.score);
    }

    let n = data.len() as f64;
    let sum: f64 = data.iter().map(|r| r.score).sum();
    let mean = sum / n;

    let variance = data
        .iter()
        .map(|r| (r.score - mean).powi(2))
        .sum::<f64>() / n;

    let std_dev = variance.sqrt();
    let median = data[data.len() / 2].score;

    println!("\nDescriptive Analytics:");
    println!("Count   : {}", n);
    println!("Mean    : {:.2}", mean);
    println!("Std Dev : {:.2}", std_dev);
    println!("Median  : {:.2}", median);

    println!("\nStandardized Scores (Z-Score):");
    for r in &data {
        let z = (r.score - mean) / std_dev;
        println!("{:<8} | Z: {:>5.2}", r.name, z);
    }
}

$d = "content/paper"
$max = (Get-ChildItem $d -Directory | Where-Object { $_.Name -match '^\d+-' } | ForEach-Object { [int]($_.Name -replace '-.*$', '') } | Measure-Object -Maximum).Maximum
if (-not $max) { $max = 0 }
$n = $max + 1
$t = Read-Host "Masukkan judul post"
$slug = $t.ToLower() -replace '[^\w\s]', '' -replace '\s+', '-'
$target = "$d/$n-$slug"
hugo new "$target/index.md"
$filePath = "$target/index.md"
if (Test-Path $filePath) {
    (Get-Content $filePath) -replace "title = '.*'", "title = '$t'" | Set-Content $filePath
}
Set-Location $target
